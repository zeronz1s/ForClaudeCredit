-- Linear lookup in a {{x, y}, ...} table, clamped at both ends.
function CarConfig.curve(points, x: number): number
	if x <= points[1][1] then
		return points[1][2]
	end
	for i = 2, #points do
		local a, b = points[i - 1], points[i]
		if x <= b[1] then
			local t = (x - a[1]) / (b[1] - a[1])
			return a[2] + (b[2] - a[2]) * t
		end
	end
	return points[#points][2]
end
